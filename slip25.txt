Q.1) Write a C Program that demonstrates redirection of standard output to a file(SLIP
7,22)
#include <stdio.h>

int main() {
    freopen("output.txt", "w", stdout); // Redirect stdout to output.txt
    printf("This message will be written to output.txt\n");
    return 0;
}

Q.2) Write a C program that redirects standard output to a file output.txt. (use of dup
and open system call).(SLIP 7,8)

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>

int main() {
    int fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    dup2(fd, STDOUT_FILENO); // Redirect stdout to file
    close(fd);               // Close original file descriptor

    printf("This message is redirected using dup and open.\n");

    return 0;
}
