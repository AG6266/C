Q.1) Write a C program that illustrates suspending and resuming processes using
signal s

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>

void handler(int sig) {
    if (sig == SIGCONT) {
        printf("Process resumed\n");
    } else if (sig == SIGTSTP) {
        printf("Process suspended\n");
    }
}

int main() {
    signal(SIGCONT, handler);
    signal(SIGTSTP, handler);

    printf("Process running. Use Ctrl+Z to suspend and 'fg' to resume.\n");

    while (1) {
        sleep(1);
    }

    return 0;
}


Q.2) Write a C program that a string as an argument and return all the
files that begins

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>

int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <prefix>\n", argv[0]);
        return 1;
    }

    DIR *dir;
    struct dirent *entry;

    dir = opendir(".");
    if (dir == NULL) {
        perror("opendir");
        return 1;
    }

    while ((entry = readdir(dir)) != NULL) {
        if (strncmp(entry->d_name, argv[1], strlen(argv[1])) == 0) {
            printf("%s\n", entry->d_name);
        }
    }

    closedir(dir);
    return 0;
}
