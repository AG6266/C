Q.1) Write a C program to find whether a given file is present in curr
ent directory or
not(SLIP 23)

#include <stdio.h>
#include <dirent.h>
#include <string.h>

int main() {
    char filename[100];
    int found = 0;

    printf("Enter filename to search: ");
    scanf("%s", filename);

    DIR *dir = opendir(".");
    if (dir == NULL) {
        perror("opendir");
        return 1;
    }

    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        if (strcmp(entry->d_name, filename) == 0) {
            found = 1;
            break;
        }
    }

    closedir(dir);

    if (found)
        printf("File '%s' found in current directory.\n", filename);
    else
        printf("File '%s' not found.\n", filename);

    return 0;
}


Q.2) Write a C program to create an unnamed pipe. The child process will write
following three
messages to pipe and parent process display it.
Message1 =
Hello World ”
Message2 =
Hello SPPU ”
Message3 =
Linux is Funny SLIP 5)

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

int main() {
    int fd[2];
    pid_t pid;
    char buffer[100];

    if (pipe(fd) == -1) {
        perror("pipe");
        return 1;
    }

    pid = fork();

    if (pid == 0) {
        // Child process
        close(fd[0]); // Close read end

        write(fd[1], "Message1 = Hello World\n", strlen("Message1 = Hello World\n"));
        write(fd[1], "Message2 = Hello SPPU\n", strlen("Message2 = Hello SPPU\n"));
        write(fd[1], "Message3 = Linux is Funny\n", strlen("Message3 = Linux is Funny\n"));

        close(fd[1]);
        exit(0);
    } else if (pid > 0) {
        // Parent process
        close(fd[1]); // Close write end

        printf("Parent reading messages from pipe:\n");
        while (read(fd[0], buffer, sizeof(buffer)) > 0) {
            printf("%s", buffer);
        }

        close(fd[0]);
        wait(NULL);
    } else {
        perror("fork");
        return 1;
    }

    return 0;
}
