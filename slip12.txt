Q.1) Write a C program that print the exit status of a terminated child process

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();

    if (pid == 0) {
        // Child process
        exit(5); // Exit with status 5
    } else if (pid > 0) {
        int status;
        wait(&status);

        if (WIFEXITED(status)) {
            printf("Child exited with status: %d\n", WEXITSTATUS(status));
        } else {
            printf("Child did not exit normally\n");
        }
    } else {
        perror("fork");
        return 1;
    }

    return 0;
}


Q.2) Write a C program which receives file names as command line arguments and
display those filenames i n ascending order according to their sizes. I)

#include <stdio.h>
#include <stdlib.h>
#include <sys/stat.h>
#include <string.h>

struct FileInfo {
    char name[256];
    long size;
};

int compare(const void *a, const void *b) {
    struct FileInfo *f1 = (struct FileInfo *)a;
    struct FileInfo *f2 = (struct FileInfo *)b;
    return f1->size - f2->size;
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Usage: %s <file1> <file2> ...\n", argv[0]);
        return 1;
    }

    struct FileInfo files[argc - 1];
    struct stat statbuf;

    for (int i = 1; i < argc; i++) {
        if (stat(argv[i], &statbuf) == 0) {
            strcpy(files[i - 1].name, argv[i]);
            files[i - 1].size = statbuf.st_size;
        } else {
            perror(argv[i]);
            files[i - 1].size = -1;
        }
    }

    qsort(files, argc - 1, sizeof(struct FileInfo), compare);

    printf("Files in ascending order of size:\n");
    for (int i = 0; i < argc - 1; i++) {
        if (files[i].size != -1)
            printf("%s (%ld bytes)\n", files[i].name, files[i].size);
    }

    return 0;
}
