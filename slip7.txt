#include <stdio.h>

int main() {
    // Redirect stdout to "output.txt"
    FILE *file = freopen("output.txt", "w", stdout);
    if (file == NULL) {
        perror("Error opening file");
        return 1;
    }

    // These lines will go into output.txt
    printf("This line will be written to the output.txt file.\n");
    printf("Standard output is now redirected to a file!\n");

    // Close the file
    fclose(file);

    // Restore stdout to terminal (UNIX systems)
    freopen("/dev/tty", "w", stdout);

    // This line will appear on the terminal
    printf("This line will be displayed on the terminal again.\n");

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    int pipe_fd[2];
    pid_t pid1, pid2;

    if (pipe(pipe_fd) == -1) {
        perror("Pipe creation failed");
        exit(EXIT_FAILURE);
    }

    pid1 = fork();
    if (pid1 == -1) {
        perror("Fork failed for first child");
        exit(EXIT_FAILURE);
    }

    if (pid1 == 0) {
        close(pipe_fd[0]);
        dup2(pipe_fd[1], STDOUT_FILENO);
        close(pipe_fd[1]);
        execlp("ls", "ls", "-l", NULL);
        perror("execlp failed for ls");
        exit(EXIT_FAILURE);
    }

    pid2 = fork();
    if (pid2 == -1) {
        perror("Fork failed for second child");
        exit(EXIT_FAILURE);
    }

    if (pid2 == 0) {
        close(pipe_fd[1]);
        dup2(pipe_fd[0], STDIN_FILENO);
        close(pipe_fd[0]);
        execlp("wc", "wc", "-l", NULL);
        perror("execlp failed for wc");
        exit(EXIT_FAILURE);
    }

    close(pipe_fd[0]);
    close(pipe_fd[1]);

    wait(NULL);
    wait(NULL);

    return 0;
}
