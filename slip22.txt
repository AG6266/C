Q.1) Write a C Program that demonstrates redirection of standard outpu
t to a file(SLIP
7,25)


#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>

int main() {
    int fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    dup2(fd, STDOUT_FILENO); // Redirect stdout to file
    close(fd);

    printf("This message goes to output.txt\n");

    return 0;
}

Q.2) Write a C program to implement the following unix/linux command (use fork, pipe
and exec system call).
Your program should block the signal Ctrl
C and Ctrl signal during the execution.
i. ls
l | wc l(SLIP 17)

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <sys/wait.h>

int main() {
    int fd[2];
    pid_t pid1, pid2;

    // Block SIGINT (Ctrl+C) and SIGQUIT (Ctrl+\)
    signal(SIGINT, SIG_IGN);
    signal(SIGQUIT, SIG_IGN);

    if (pipe(fd) == -1) {
        perror("pipe");
        exit(1);
    }

    pid1 = fork();
    if (pid1 == 0) {
        dup2(fd[1], STDOUT_FILENO);
        close(fd[0]);
        close(fd[1]);
        execlp("ls", "ls", "-l", NULL);
        perror("execlp ls");
        exit(1);
    }

    pid2 = fork();
    if (pid2 == 0) {
        dup2(fd[0], STDIN_FILENO);
        close(fd[1]);
        close(fd[0]);
        execlp("wc", "wc", "-l", NULL);
        perror("execlp wc");
        exit(1);
    }

    close(fd[0]);
    close(fd[1]);

    wait(NULL);
    wait(NULL);

    return 0;
}
